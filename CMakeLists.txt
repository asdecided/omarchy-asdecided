cmake_minimum_required(VERSION 3.21)
project(AsDecided VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup()
qt_add_executable(asdecided app/main.cpp app/MarkdownView.cpp app/MarkdownView.h app/Backend.cpp app/Workspace.cpp app/Backend.h)
qt_add_resources(asdecided "qml" PREFIX "/" FILES app/qml/Main.qml app/qml/Theme.qml)
target_link_libraries(asdecided PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_options(asdecided PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS asdecided RUNTIME DESTINATION bin)
install(PROGRAMS "${CMAKE_SOURCE_DIR}/target/release/asdecided-desktop-backend" DESTINATION bin)
install(FILES packaging/io.github.asdecided.AsDecided.desktop DESTINATION share/applications)
install(FILES packaging/io.github.asdecided.AsDecided.png DESTINATION share/icons/hicolor/64x64/apps)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
    qt_add_executable(bridge-test tests/bridge_test.cpp app/Backend.cpp app/Workspace.cpp app/Backend.h)
    target_include_directories(bridge-test PRIVATE app)
    target_link_libraries(bridge-test PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
    target_compile_definitions(bridge-test PRIVATE FIXTURE_ROOT="${CMAKE_SOURCE_DIR}/tests/fixture")
    add_test(NAME bridge COMMAND bridge-test)
    set_tests_properties(bridge PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ASDECIDED_BACKEND=${CMAKE_SOURCE_DIR}/target/debug/asdecided-desktop-backend")
endif()
